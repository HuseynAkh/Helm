# Opening and closing folders in the list moves rows into place and never rebuilds the list. Works in its own tab,
# so the tab that's saved at quit is left as it was.
cmd newTab
wait 0.3
go @FIX@/home/Unfold
view list
sort-choose Name A–Z
wait 0.3
list-rows Many,Nested,top.txt
select top.txt
list-mark
# A first opening: the rows arrive under the folder, in order, and the selection stays where it was
disclose Many
list-open Many
list-in-step
child Many/file-01.txt/file-40.txt
list-selected top.txt
selected top.txt
disclose Many
list-closed Many
list-in-step
list-rows Many,Nested,top.txt
selected top.txt
# Opening it again shows its rows from the listing it already has
disclose Many
list-in-step
child file-01.txt/file-40.txt
disclose Many
list-in-step
# A folder inside a folder comes back open with its parent

disclose Inner

disclose Nested
list-in-step
list-rows Many,Nested,top.txt
disclose Nested

list-open Inner
list-rows Many,Nested,Inner,deep.txt,n.txt,top.txt
