# Properties: a folder's contents counted, with what couldn't be read, its parent folder and Nautilus's permissions,
# which the owner changes with Finder's pop-ups.
# In a tab of its own, closed at the end, so the steps after these find the window as it was.
cmd newTab
wait 0.3
go @FIX@/home
select Measured
cmd properties
properties-wait
properties-windows 1
properties-title Measured Properties
properties-row Contents=8 items, totalling 45 bytes
properties-note Contents=1 folder inside couldn’t be read, so what’s in it isn’t counted.
properties-row Parent Folder=~
properties-row Owner=You
properties-note Owner=Can create and delete files
properties-note Others=Can access files
properties-access Others=Read only
properties-note Disk=GB free of
properties-row Locked=No
properties-lacks Size
properties-lacks Runs as Program
properties-shot @SHOTS@/properties-folder.png
properties-close escape
properties-windows 0

# Permissions the owner changes, as in Finder's Get Info and Nautilus's Permissions page. Each change applies at once,
# is read back from the disk and is one change in the window's history, which Undo and Redo in the Properties window
# or the browser window take back and make again.
go @FIX@/home/Access
select notes.txt
cmd properties
properties-wait
properties-row Owner=You
properties-access Owner=Read & Write
properties-access Group=Read only
properties-access Others=Read only
properties-note Group=Can only read
properties-execute off
properties-controls enabled
properties-footer
properties-lacks Runs as Program
properties-shot @SHOTS@/properties-permissions.png
properties-choose Group=No access
fixture-mode 0604 @FIX@/home/Access/notes.txt
properties-access Group=No access
properties-note Group=No access
properties-footer
properties-undo-title Undo Change Permissions
# The checkbox lets each class that can read the file run it, and off lets nobody
properties-execute-set on
fixture-mode 0705 @FIX@/home/Access/notes.txt
properties-execute on
properties-undo
fixture-mode 0604 @FIX@/home/Access/notes.txt
properties-execute off
properties-undo
fixture-mode 0644 @FIX@/home/Access/notes.txt
properties-access Group=Read only
properties-redo
fixture-mode 0604 @FIX@/home/Access/notes.txt
properties-access Group=No access
properties-close
# The browser window's Undo takes back a change made in Properties, as one of its own
undo-title Change Permissions
cmd undo
fixture-mode 0644 @FIX@/home/Access/notes.txt
