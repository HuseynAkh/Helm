# Properties: a folder's contents counted, with what couldn't be read, its parent folder and Nautilus's permissions
go @FIX@/home
view list
select Measured
cmd properties
properties-wait
properties-windows 1
properties-title Measured Properties
properties-row Contents=8 items, totalling 45 bytes
properties-note Contents=1 folder inside couldn’t be read, so what’s in it isn’t counted.
properties-row Parent Folder=~
properties-row Owner=You
properties-note Owner=Can create and delete files
properties-row Others=Can access files
properties-note Disk=GB free of
properties-row Locked=No
properties-lacks Size
properties-lacks Runs as Program
properties-shot @SHOTS@/properties-folder.png
properties-close escape
properties-windows 0
# A file: its size once, its type as Nautilus words it, and whether it runs
go @FIX@/home/Measured
select a.txt
cmd properties
properties-wait
properties-row Size=5 bytes
properties-row Type=text/plain
properties-row Runs as Program=No
properties-row Parent Folder=~/Measured
properties-note Owner=Can read and write
properties-lacks Disk
properties-lacks Contents
properties-shot @SHOTS@/properties-file.png
properties-close
# A package counts what's inside it as its size

# Several items: how many of each, their size together, and what's inside the folders

properties-row Size=35 bytes
properties-note Size=With 3 items inside the folders.
